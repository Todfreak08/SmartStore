// ========================================================== // AUTHENTICATION + LOGIN ACTIVITY // ========================================================== auth.onAuthStateChanged((user) => { if (!user) { window.location.href = "index.html"; return; } const userEmail = document.getElementById("userEmail"); if (userEmail) { userEmail.innerHTML = user.email; } // Save login activity recordLoginActivity(user); }); // ========================================================== // RECORD LOGIN ACTIVITY // ========================================================== function recordLoginActivity(user) { const loginData = { email: user.email || "Unknown", uid: user.uid || "Unknown", action: "LOGIN", timestamp: new Date().toLocaleString(), createdAt: firebase.database .ServerValue .TIMESTAMP, userAgent: navigator.userAgent }; database .ref("SmartStorage/loginActivity") .push(loginData) .then(() => { console.log( "✅ Login activity saved to Firebase." ); }) .catch((error) => { console.error( "❌ Failed to save login activity:", error ); }); } // ========================================================== // LOGOUT // ========================================================== function logout() { const user = auth.currentUser; if (user) { const logoutData = { email: user.email || "Unknown", uid: user.uid || "Unknown", action: "LOGOUT", timestamp: new Date().toLocaleString(), createdAt: firebase.database .ServerValue .TIMESTAMP }; database .ref( "SmartStorage/loginActivity" ) .push(logoutData) .then(() => { return auth.signOut(); }) .then(() => { window.location.href = "index.html"; }) .catch((error) => { console.error( "Logout error:", error ); }); } else { window.location.href = "index.html"; } }
